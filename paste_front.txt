function handleTagInput(e){
  if(['Enter','Tab',','].includes(e.key)){
    e.preventDefault();
    const val = e.target.value.trim().toLowerCase().replace(/[^a-z0-9\-]/g,'');
    if(val && !tags.includes(val) && tags.length < 10){
      tags.push(val);
      renderTags();
    }
    e.target.value='';
  }
  if(e.key==='Backspace' && !e.target.value && tags.length){
    tags.pop();renderTags();
  }
}
function renderTags(){
  const wrap = document.getElementById('tagsWrap');
  const input = document.getElementById('tagInput');
  wrap.innerHTML='';
  tags.forEach((t,i)=>{
    const chip = document.createElement('div');
    chip.className='tag-chip';
    chip.innerHTML=`${t}<button onclick="removeTag(${i})">×</button>`;
    wrap.appendChild(chip);
  });
  wrap.appendChild(input);
}
function removeTag(i){ tags.splice(i,1); renderTags(); }

async function doLogin(){
  const u = document.getElementById('loginUsername').value.trim();
  const p = document.getElementById('loginPassword').value;
  const errEl = document.getElementById('loginError');
  errEl.innerHTML='';
  if(!u||!p){ errEl.innerHTML='<div class="error-msg">Fill in all fields</div>'; return; }
  try{
    const r = await fetch('/api/auth/login',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({username:u,password:p})});
    const d = await r.json();
    if(!r.ok){ errEl.innerHTML=`<div class="error-msg">${d.detail}</div>`; return; }
    closeAuthModal();
    await checkAuth();
    showToast('Welcome back, '+u+'!','success');
  }catch(e){ errEl.innerHTML='<div class="error-msg">Connection error</div>'; }
}

}

async function createPaste(){
  const content = getEditorContent();
  if(!content.trim()){ showToast('Content cannot be empty','error'); return; }
  const body = {
    title: document.getElementById('pasteTitle').value.trim() || 'Untitled',
    content,
    language: document.getElementById('langSelect').value,
    format: currentFormat,
    visibility: document.getElementById('optVisibility').value,
    password: document.getElementById('optPassword').value || null,
    expiration: document.getElementById('optExpiry').value,
    burn_after_read: document.getElementById('burnToggle').classList.contains('on'),
    tags: [...tags]
  };
  document.getElementById('submitError').innerHTML='';
  try{
    const r = await fetch('/api/paste',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify(body)});
    const d = await r.json();
    if(!r.ok){ document.getElementById('submitError').innerHTML=`<div class="error-msg">${d.detail}</div>`; return; }
    showToast('Paste created!','success');
    viewPaste(d.short_code);
  }catch(e){ document.getElementById('submitError').innerHTML='<div class="error-msg">Server error</div>'; }
}

async function viewPaste(code){
  showPage('view');
  const wrap = document.getElementById('pasteViewContent');
  wrap.innerHTML='<div style="text-align:center;padding:80px;color:var(--text3)">Loading…</div>';
  try{
    const r = await fetch('/api/paste/'+code);
    if(r.status===404){ wrap.innerHTML='<div class="empty-state"><div class="empty-icon">🔍</div><h3>Paste Not Found</h3><p>It may have expired or been deleted.</p></div>'; return; }
    if(r.status===403){ wrap.innerHTML='<div class="empty-state"><div class="empty-icon">🔒</div><h3>Private Paste</h3><p>This paste is private.</p></div>'; return; }
    const d = await r.json();
    if(d.locked){ renderPasswordGate(d,code); }
    else { renderPasteView(d,code); }
  }catch(e){ wrap.innerHTML='<div class="empty-state"><div class="empty-icon">⚠️</div><h3>Error loading paste</h3></div>'; }
}

async function unlockPaste(code){
  const pw = document.getElementById('unlockPass').value;
  if(!pw){ showToast('Enter a password','error'); return; }
  const r = await fetch('/api/paste/'+code+'/unlock',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({password:pw})});
  if(r.status===401){ document.getElementById('unlockError').innerHTML='<div class="error-msg">Wrong password</div>'; return; }
  const d = await r.json();
  renderPasteView(d,code);
}

async function copyPasteContent(code){
  try{
    await navigator.clipboard.writeText(window._currentPasteContent||'');
    showCopyOverlay();
  }catch(e){ showToast('Copy failed','error'); }
}

async function forkPaste(code){
  const content = window._currentPasteContent;
  if(!content) return;
  setEditorContent(content);
  showPage('home');
  showToast('Content forked into editor!','info');
}

async function deletePaste(code){
  if(!confirm('Delete this paste? This cannot be undone.')) return;
  const r = await fetch('/api/paste/'+code,{method:'DELETE'});
  if(r.ok){ showToast('Paste deleted','info'); showPage('home'); }
  else showToast('Delete failed','error');
}

function downloadPaste(code, title, lang){
  const content = window._currentPasteContent||'';
  const ext = {'python':'py','javascript':'js','typescript':'ts','html':'html','css':'css','json':'json','java':'java','cpp':'cpp','c':'c','go':'go','rust':'rs','ruby':'rb','php':'php','sql':'sql','bash':'sh','yaml':'yml','markdown':'md'}[lang]||'txt';
  const blob = new Blob([content],{type:'text/plain'});
  const a = document.createElement('a');
  a.href=URL.createObjectURL(blob);
  a.download=(title||'paste')+'.'+ext;
  a.click();
}

function copyShareLink(code){
  const url = window.location.origin+'/view/'+code;
  navigator.clipboard.writeText(url).then(()=>showCopyOverlay()).catch(()=>showToast('Copy failed','error'));
}

function handleRoute(){
  const path = window.location.pathname;
  if(path === '/' || path === '/new' || path === '') { showPage('home'); return; }
  if(path === '/public') { showPage('public'); loadPublicPastes(1); return; }
  const viewMatch = path.match(/^\/(?:view\/)?([A-Za-z0-9]{4,12})$/);
  if(viewMatch){ viewPaste(viewMatch[1]); return; }
  const profileMatch = path.match(/^\/u\/(.+)$/);
  if(profileMatch){ showProfile(decodeURIComponent(profileMatch[1])); return; }
}

window.addEventListener('popstate', handleRoute);

// Override nav clicks to use history
document.querySelectorAll('[onclick*="showPage"]').forEach(el=>{

  el.addEventListener('click', ()=>{
    const m = orig.match(/showPage\('(\w+)'\)/);
    if(m && m[1]==='home') history.pushState({},'','/');
    else if(m && m[1]==='public') history.pushState({},'','/public');
  });
});

function escHtml(s){
  return String(s||'').replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;');
}
function formatTimeAgo(ts, future=false){
  if(!ts) return 'Never';
  const now = Math.floor(Date.now()/1000);
  const diff = future ? ts-now : now-ts;
  if(diff < 0 && !future) return 'just now';
  if(diff < 60) return diff+'s '+(future?'left':'ago');
  if(diff < 3600) return Math.floor(diff/60)+'m '+(future?'left':'ago');
  if(diff < 86400) return Math.floor(diff/3600)+'h '+(future?'left':'ago');
  if(diff < 2592000) return Math.floor(diff/86400)+'d '+(future?'left':'ago');
  if(diff < 31536000) return Math.floor(diff/2592000)+'mo '+(future?'left':'ago');
  return Math.floor(diff/31536000)+'y '+(future?'left':'ago');
}
function langIcon(lang){
  const m={'python':'🐍','javascript':'⚡','typescript':'📘','html':'🌐','css':'🎨','java':'☕','go':'🐹','rust':'🦀','sql':'🗄','json':'📦','bash':'🖥','ruby':'💎','php':'🐘','cpp':'⚙️','c':'⚙️','markdown':'📝'};
  return m[lang]||'📄';
}
